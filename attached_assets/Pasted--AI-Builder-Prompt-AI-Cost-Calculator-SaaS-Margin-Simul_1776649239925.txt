## PRICING DATA LAYER (Build This First)

### Primary Source
```
GET https://openrouter.ai/api/v1/models
```
Returns a JSON array. Each model object has:
```json
{
  "id": "openai/gpt-4o",
  "name": "OpenAI: GPT-4o",
  "context_length": 128000,
  "pricing": {
    "prompt": "0.000005",       // USD per input token
    "completion": "0.000015",   // USD per output token
    "input_cache_read": "0.0000005",
    "input_cache_write": "0.0000000"
  }
}
```
**Important:** Prices are in USD **per token**. Multiply by 1,000,000 for display as "per million tokens."

### Normalization Function
Create `lib/pricing.ts`:
```typescript
export interface ModelPrice {
  id: string;
  name: string;
  provider: string;
  inputPricePerMillion: number;    // USD
  outputPricePerMillion: number;   // USD
  cacheReadPricePerMillion: number | null;
  cacheWritePricePerMillion: number | null;
  contextLength: number;
  tokenizer: 'cl100k_base' | 'o200k_base' | 'claude' | 'sentencepiece' | 'unknown';
}

export async function fetchModels(): Promise<ModelPrice[]> {
  const res = await fetch('https://openrouter.ai/api/v1/models');
  const data = await res.json();
  return data.data
    .filter((m: any) => parseFloat(m.pricing?.prompt) > 0)
    .map((m: any) => ({
      id: m.id,
      name: m.name,
      provider: m.id.split('/')[0],
      inputPricePerMillion: parseFloat(m.pricing.prompt) * 1_000_000,
      outputPricePerMillion: parseFloat(m.pricing.completion) * 1_000_000,
      cacheReadPricePerMillion: m.pricing.input_cache_read
        ? parseFloat(m.pricing.input_cache_read) * 1_000_000 : null,
      cacheWritePricePerMillion: m.pricing.input_cache_write
        ? parseFloat(m.pricing.input_cache_write) * 1_000_000 : null,
      contextLength: m.context_length,
      tokenizer: inferTokenizer(m.id),
    }))
    .sort((a: ModelPrice, b: ModelPrice) => a.inputPricePerMillion - b.inputPricePerMillion);
}

function inferTokenizer(modelId: string): ModelPrice['tokenizer'] {
  if (modelId.includes('gpt-4o') || modelId.includes('gpt-5')) return 'o200k_base';
  if (modelId.includes('gpt-4') || modelId.includes('gpt-3.5')) return 'cl100k_base';
  if (modelId.includes('claude')) return 'claude';
  if (modelId.includes('gemini') || modelId.includes('llama') || modelId.includes('mistral')) return 'sentencepiece';
  return 'unknown';
}
```

### Fallback Static Data
Bundle a `public/model-prices-fallback.json` with the top 20 models hardcoded. Used if the OpenRouter fetch fails.

## CALCULATION ENGINE

Create `lib/calculator.ts` with these exact functions. Do not alter the formulas.

```typescript
// ── Core cost functions ──────────────────────────────────────────────────────

export function calcCallCost(
  inputTokens: number,
  outputTokens: number,
  model: ModelPrice
): number {
  return (
    (inputTokens * model.inputPricePerMillion) / 1_000_000 +
    (outputTokens * model.outputPricePerMillion) / 1_000_000
  );
}

export function calcMonthlyCost(
  callCostUSD: number,
  callsPerDay: number
): number {
  return callCostUSD * callsPerDay * 30;
}

// ── SaaS margin functions ────────────────────────────────────────────────────

export interface MarginResult {
  mrr: number;
  aiCOGS: number;
  otherCOGS: number;
  grossProfit: number;
  grossMarginPct: number;
  cpau: number;                  // cost per active user
  revenuePerUser: number;
  contributionMarginPerUser: number;
}

export function calcGrossMargin(
  mau: number,
  arpu: number,                  // average revenue per user per month
  callsPerUserPerDay: number,
  avgInputTokens: number,
  avgOutputTokens: number,
  model: ModelPrice,
  otherCOGSPerUser: number = 1.50  // default: $1.50/user/mo (hosting + Stripe)
): MarginResult {
  const callCost = calcCallCost(avgInputTokens, avgOutputTokens, model);
  const cpau = callCost * callsPerUserPerDay * 30;
  const mrr = mau * arpu;
  const aiCOGS = mau * cpau;
  const otherCOGS = mau * otherCOGSPerUser;
  const grossProfit = mrr - aiCOGS - otherCOGS;
  return {
    mrr,
    aiCOGS,
    otherCOGS,
    grossProfit,
    grossMarginPct: mrr > 0 ? (grossProfit / mrr) * 100 : 0,
    cpau,
    revenuePerUser: arpu,
    contributionMarginPerUser: arpu - cpau - otherCOGSPerUser,
  };
}

// ── Breakeven function ───────────────────────────────────────────────────────

export function calcBreakevenMAU(
  fixedCostsPerMonth: number,
  arpu: number,
  cpau: number,
  otherCOGSPerUser: number = 1.50
): number {
  const contributionMargin = arpu - cpau - otherCOGSPerUser;
  if (contributionMargin <= 0) return Infinity;
  return Math.ceil(fixedCostsPerMonth / contributionMargin);
}

// ── Power user risk function ─────────────────────────────────────────────────

export interface PowerUserResult {
  blendedCPAU: number;
  lightUserCost: number;
  avgUserCost: number;
  heavyUserCost: number;
  upliftVsFlatPct: number;       // how much higher blended is vs naive flat CPAU
  top10PctMonthlyBurden: number; // total cost from top 10% users at given MAU
}

export function calcPowerUserRisk(
  baseCPAU: number,
  mau: number,
  lightPct: number = 0.60,
  avgPct: number = 0.30,
  heavyPct: number = 0.10,
  lightMultiplier: number = 0.33,
  avgMultiplier: number = 1.67,
  heavyMultiplier: number = 3.00
): PowerUserResult {
  const lightUserCost = baseCPAU * lightMultiplier;
  const avgUserCost = baseCPAU * avgMultiplier;
  const heavyUserCost = baseCPAU * heavyMultiplier;
  const blendedCPAU =
    lightPct * lightUserCost +
    avgPct * avgUserCost +
    heavyPct * heavyUserCost;
  return {
    blendedCPAU,
    lightUserCost,
    avgUserCost,
    heavyUserCost,
    upliftVsFlatPct: ((blendedCPAU - baseCPAU) / baseCPAU) * 100,
    top10PctMonthlyBurden: mau * heavyPct * heavyUserCost,
  };
}

// ── Prompt caching savings function ─────────────────────────────────────────

export function calcCachingSavings(
  sysPromptTokens: number,
  callsPerDay: number,
  model: ModelPrice
): { monthlySavings: number; savingsPct: number } {
  if (!model.cacheReadPricePerMillion) return { monthlySavings: 0, savingsPct: 0 };
  const withoutCaching =
    (sysPromptTokens * model.inputPricePerMillion) / 1_000_000;
  const withCaching =
    (sysPromptTokens * model.cacheReadPricePerMillion) / 1_000_000;
  const savingsPerCall = withoutCaching - withCaching;
  // First call writes cache (no saving), subsequent calls save
  const totalCalls = callsPerDay * 30;
  const monthlySavings = savingsPerCall * (totalCalls - 1);
  const savingsPct = withoutCaching > 0
    ? (savingsPerCall / withoutCaching) * 100 : 0;
  return { monthlySavings, savingsPct };
}
```
